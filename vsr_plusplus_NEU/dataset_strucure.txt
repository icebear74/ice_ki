# NEW Dataset Structure (dataset_generator_v2)
# Base: /mnt/data/training/datasetNeu

# Categories (lowercase): master, universal, space, toon
# Each category has size-specific subdirectories

# Example for 'master' category:
/mnt/data/training/datasetNeu/master/
├── patches/
│   ├── 540/
│   │   ├── GT/           # 540×540 Ground Truth
│   │   └── LR_7frames/   # 180×1260 (7 frames horizontal)
│   ├── 720/
│   │   ├── GT/           # 720×720 Ground Truth
│   │   └── LR_7frames/   # 240×1680 (7 frames horizontal)
│   └── 720_169/
│       ├── GT/           # 405×720 Ground Truth (16:9)
│       └── LR_7frames/   # 135×1680 (7 frames horizontal)
│
└── val/                  # Validation (manual setup)
    ├── 540/
    │   └── GT/           # Only GT needed (LR auto-found in patches)
    ├── 720/
    │   └── GT/
    └── 720_169/
        └── GT/
